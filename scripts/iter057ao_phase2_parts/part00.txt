def q(seed,*idx,mod=11):
    b=(seed+':' + ':'.join(map(str,idx))).encode()
    n=int(hashlib.sha256(b).hexdigest()[:16],16)
    return F((n%(2*mod+1))-mod,13)


def gderiv(seed,a,b,ders):
    if a>b: a,b=b,a
    ds=tuple(sorted(ders)); k=len(ds)
    if k==0:
        base=F(-1 if a==0 else 1) if a==b else F(0)
        return base + q(seed,'g',a,b,mod=2)/20
    if k==1: return q(seed,'dg',a,b,*ds,mod=4)/7
    if k==2: return q(seed,'ddg',a,b,*ds,mod=4)/9
    if k==3: return q(seed,'dddg',a,b,*ds,mod=4)/11
    if k==4: return q(seed,'ddddg',a,b,*ds,mod=4)/13
    raise ValueError(k)


def hderiv(seed,a,b,ders):
    if a>b:a,b=b,a
    ds=tuple(sorted(ders)); k=len(ds)
    if k==0:return q(seed,'h',a,b,mod=5)
    if k==1:return q(seed,'dh',a,b,*ds,mod=4)/7
    if k==2:return q(seed,'ddh',a,b,*ds,mod=4)/9
    raise ValueError(k)


class J2:
    __slots__=('v','d','dd')
    def __init__(self,v=F(0),d=None,dd=None):
        self.v=F(v)
        self.d=[F(0)]*N if d is None else [F(x) for x in d]
        self.dd=[[F(0)]*N for _ in range(N)] if dd is None else [[F(x) for x in r] for r in dd]
    @staticmethod
    def c(x): return x if isinstance(x,J2) else J2(x)
    def __add__(self,o):
        o=J2.c(o)
        return J2(self.v+o.v,[self.d[i]+o.d[i] for i in range(N)],[[self.dd[i][j]+o.dd[i][j] for j in range(N)] for i in range(N)])
    __radd__=__add__
    def __neg__(self): return J2(-self.v,[-x for x in self.d],[[-x for x in r] for r in self.dd])
    def __sub__(self,o): return self+(-J2.c(o))
    def __rsub__(self,o): return J2.c(o)+(-self)
    def __mul__(self,o):
        o=J2.c(o); v=self.v*o.v
        d=[self.d[i]*o.v+self.v*o.d[i] for i in range(N)]
        dd=[[self.dd[i][j]*o.v+self.d[i]*o.d[j]+self.d[j]*o.d[i]+self.v*o.dd[i][j] for j in range(N)] for i in range(N)]
        return J2(v,d,dd)
    __rmul__=__mul__
    def scale(self,c): return self*F(c)


def matmul(A,B):
    n=len(A); m=len(B[0]); k=len(B)
    return [[sum((A[i][r]*B[r][j] for r in range(k)),J2()) for j in range(m)] for i in range(n)]


def matmulF(A,B):
    n=len(A);m=len(B[0]);k=len(B)
    return [[sum((A[i][r]*B[r][j] for r in range(k)),F(0)) for j in range(m)] for i in range(n)]


def mataddF(A,B,scaleB=1):
    return [[A[i][j]+scaleB*B[i][j] for j in range(len(A[0]))] for i in range(len(A))]


def inv_fraction(A):
    n=len(A); aug=[list(map(F,A[i]))+[F(int(i==j)) for j in range(n)] for i in range(n)]
    det=F(1); sign=1
    for c in range(n):
        p=next((r for r in range(c,n) if aug[r][c]),None)
        if p is None: raise ZeroDivisionError('singular')
        if p!=c: aug[c],aug[p]=aug[p],aug[c]; sign*=-1
        pv=aug[c][c]; det*=pv
        aug[c]=[x/pv for x in aug[c]]
        for r in range(n):
            if r==c: continue
            z=aug[r][c]
            if z: aug[r]=[aug[r][j]-z*aug[c][j] for j in range(2*n)]
    return [row[n:] for row in aug], det*sign


def background_from_deriv(fn):
    g0=[[fn(a,b,()) for b in range(N)] for a in range(N)]
    gi0,det=inv_fraction(g0)
    dg=[[[fn(a,b,(c,)) for c in range(N)] for b in range(N)] for a in range(N)]
    ddg=[[[[fn(a,b,(c,d)) for d in range(N)] for c in range(N)] for b in range(N)] for a in range(N)]
    # inverse derivatives
    DG=[]
    for c in range(N): DG.append([[dg[a][b][c] for b in range(N)] for a in range(N)])
    DDG=[[None]*N for _ in range(N)]
    for c in range(N):
      for d in range(N): DDG[c][d]=[[ddg[a][b][c][d] for b in range(N)] for a in range(N)]
    def negmat(M): return [[-x for x in r] for r in M]
    gid=[]
    for c in range(N): gid.append(negmat(matmulF(matmulF(gi0,DG[c]),gi0)))
    gidd=[[None]*N for _ in range(N)]
    for c in range(N):
      for d in range(N):
        t1=matmulF(matmulF(matmulF(matmulF(gi0,DG[d]),gi0),DG[c]),gi0)
        t2=matmulF(matmulF(matmulF(matmulF(gi0,DG[c]),gi0),DG[d]),gi0)
        t3=matmulF(matmulF(gi0,DDG[c][d]),gi0)
        gidd[c][d]=mataddF(mataddF(t1,t2),t3,-1)
    G=[[J2(g0[a][b],[dg[a][b][c] for c in range(N)],[[ddg[a][b][c][d] for d in range(N)] for c in range(N)]) for b in range(N)] for a in range(N)]
    Gi=[[J2(gi0[a][b],[gid[c][a][b] for c in range(N)],[[gidd[c][d][a][b] for d in range(N)] for c in range(N)]) for b in range(N)] for a in range(N)]
    D=[[[J2(fn(a,b,(c,)),[fn(a,b,(c,e)) for e in range(N)],[[fn(a,b,(c,e,f)) for f in range(N)] for e in range(N)]) for c in range(N)] for b in range(N)] for a in range(N)]
    DD=[[[[J2(fn(a,b,(c,d)),[fn(a,b,(c,d,e)) for e in range(N)],[[fn(a,b,(c,d,e,f)) for f in range(N)] for e in range(N)]) for d in range(N)] for c in range(N)] for b in range(N)] for a in range(N)]
    return {'g0':g0,'gi0':gi0,'det':det,'G':G,'Gi':Gi,'D':D,'DD':DD,'fn':fn}
