def eh_compact_formula_control(bg,H,Hd):
    gi=bg['gi0'];R=bg['R'];Ric=bg['Ric'];Sc=bg['Scal'].v;Gam=bg['Gam']
    bulk_ok=True
    for a,b in product(range(N),repeat=2):
        T=F(0)
        for c,d,e in product(range(N),repeat=3):
            P=F(1,2)*(gi[a][d]*gi[c][e]-gi[a][e]*gi[c][d])
            Rub=sum((gi[b][f]*R[f][c][d][e].v for f in range(N)),F(0))
            T+=P*Rub
        Ricup=sum((gi[a][i]*gi[b][j]*Ric[i][j].v for i,j in product(range(N),repeat=2)),F(0))
        lhs=-T+F(1,2)*gi[a][b]*Sc
        rhs=-Ricup+F(1,2)*gi[a][b]*Sc
        bulk_ok=bulk_ok and lhs==rhs
    # EH boundary: 2 P^{mu b c d} nabla_d h_bc = nabla_nu h^{mu nu}-nabla^mu h.
    Gup=[[[F(0)]*N for _ in range(N)] for __ in range(N)]
    for a,b,c in product(range(N),repeat=3):Gup[a][b][c]=sum((gi[a][e]*Gam[e][b][c].v for e in range(N)),F(0))
    nab=[[[F(0)]*N for _ in range(N)] for __ in range(N)]
    for b,c,d in product(range(N),repeat=3):
        nab[b][c][d]=Hd[b][c][d]-sum((Gup[e][d][b]*H[e][c]+Gup[e][d][c]*H[b][e] for e in range(N)),F(0))
    boundary_ok=True
    for mu in range(N):
        th=F(0)
        for b,c,d in product(range(N),repeat=3):
            P=F(1,2)*(gi[mu][c]*gi[b][d]-gi[mu][d]*gi[b][c])
            th+=2*P*nab[b][c][d]
        rhs1=sum((gi[mu][a]*gi[nu][b]*nab[a][b][nu] for a,nu,b in product(range(N),repeat=3)),F(0))
        rhs2=sum((gi[mu][nu]*gi[a][b]*nab[a][b][nu] for nu,a,b in product(range(N),repeat=3)),F(0))
        boundary_ok=boundary_ok and th==rhs1-rhs2
    return {'EH_bulk_reduces_to_minus_Einstein_for_covariant_h':bulk_ok,'EH_boundary_reduces_to_standard_theta':boundary_ok}
